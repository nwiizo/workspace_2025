### 核心実装（120行）

**1. AST解析で結合パターンを検出**

```rust
use syn::visit::Visit;

pub struct CouplingAnalyzer {
    pub metrics: CouplingMetrics,
    pub current_module: String,
}

impl<'ast> Visit<'ast> for CouplingAnalyzer {
    fn visit_item_impl(&mut self, node: &'ast syn::ItemImpl) {
        if node.trait_.is_some() {
            self.metrics.contract_count += 1;  // トレイト実装 = 弱い結合
        } else {
            self.metrics.intrusive_count += 1;  // 具象型実装 = 強い結合
        }
    }
}
```

**2. Git履歴から変動性を分析**

```rust
pub fn analyze_git_history(&mut self, months: usize) -> Result<(), VolatilityError> {
    let output = Command::new("git")
        .args(["log", "--pretty=format:", "--name-only",
               &format!("--since={} months ago", months)])
        .output()?;

    let files = String::from_utf8(output.stdout)?;
    for file in files.lines().filter(|l| !l.is_empty()) {
        *self.file_changes.entry(file.to_string()).or_insert(0) += 1;
    }
    Ok(())
}
```

**3. バランススコアを計算**

```rust
pub fn calculate_balance(&self) -> f64 {
    // 強結合×近距離 または 弱結合×遠距離 = 良い
    let modularity = (self.strength * (1.0 - self.distance))
                   + ((1.0 - self.strength) * self.distance);

    // モジュラーか安定している = バランスが良い
    let stability = 1.0 - self.volatility;
    modularity.max(stability)
}
```

**4. 問題を自動検出**

```rust
pub fn identify_issues(&self) -> Vec<CouplingIssue> {
    let mut issues = Vec::new();

    // パターン1: グローバル複雑性（強結合×遠距離）
    if self.strength > 0.7 && self.distance > 0.7 {
        issues.push(CouplingIssue {
            severity: Critical,
            description: "強い結合が長距離に渡っています",
            recommendation: "コンポーネントを近づけるか、結合を弱めてください",
        });
    }

    // パターン2: カスケード変更リスク（強結合×高変動）
    if self.strength > 0.7 && self.volatility > 0.7 {
        issues.push(CouplingIssue {
            severity: Critical,
            description: "頻繁に変更されるコンポーネントに強結合しています",
            recommendation: "変動性の高いコンポーネントを隔離してください",
        });
    }

    issues
}
```
